Complete Blueprint: Building & Training a Custom System One Model
ModernBERT Backbone + RLCD Loss

This complete PyTorch script defines the custom decision head, gathers marker coordinate embeddings, applies strictly proper scoring rules (RLCD), and trains the model in two phases (Head Warmup → Joint End-to-End Fine-Tuning).

train_system_one.py (PyTorch & HuggingFace ModernBERT) 
import torch
import torch.nn as nn
from transformers import AutoModel, AutoTokenizer

# 1. THE COMPLETE SYSTEM ONE DECISION HEAD
class CustomSystemOneHead(nn.Module):
    def __init__(self, hidden_dim: int = 1024, head_layers: int = 2, dropout: float = 0.1):
        super().__init__()
        self.hidden_dim = hidden_dim
        # Question type embedding: 0 = choice, 1 = score, 2 = noul
        self.type_emb = nn.Embedding(3, hidden_dim)
        
        # 2-layer Pre-Norm Transformer Head
        nhead = max(1, hidden_dim // 64)
        layer = nn.TransformerEncoderLayer(
            d_model=hidden_dim, nhead=nhead, dim_feedforward=4 * hidden_dim,
            dropout=dropout, batch_first=True, norm_first=True
        )
        self.transformer_head = nn.TransformerEncoder(layer, num_layers=head_layers)
        
        # Marker Scorer MLP
        self.scorer = nn.Sequential(
            nn.LayerNorm(hidden_dim),
            nn.Linear(hidden_dim, hidden_dim),
            nn.GELU(),
            nn.Linear(hidden_dim, 1)
        )
        
        # Act Head for Selective Abstention / Guardrail
        self.act_head = nn.Sequential(
            nn.Linear(hidden_dim + 4, 256),
            nn.GELU(),
            nn.Linear(256, 2)
        )

    def forward(self, h_enc, attention_mask, marker_positions, marker_mask, qtype):
        # Add question type inductive bias
        h = h_enc + self.type_emb(qtype)[:, None, :]
        pad_mask = ~attention_mask.bool()
        h = self.transformer_head(h, src_key_padding_mask=pad_mask)
        
        # Gather vectors at exact [MASK] marker coordinates: [B, K, D]
        idx = marker_positions.clamp(min=0)[:, :, None].expand(-1, -1, self.hidden_dim)
        marker_vectors = torch.gather(h, 1, idx)
        
        # Compute scalar scores for each marker
        logits = self.scorer(marker_vectors).squeeze(-1).float()
        logits = logits.masked_fill(~marker_mask, -1e4)
        return logits

# 2. STRICTLY PROPER SCORING RULE LOSS (RLCD)
def rlcd_proper_reward(probs, target, qtype, mask, w_sph=0.5, w_rps=1.0):
    probs = probs * mask
    log_probs = torch.log(probs.clamp_min(1e-12)).clamp_min(-9.21)
    
    # Logarithmic score (Cross-Entropy)
    log_score = (target * log_probs).sum(-1)
    
    # Spherical score (Balances extreme gradients)
    spherical_score = (target * probs).sum(-1) / probs.norm(dim=-1).clamp_min(1e-9)
    reward = log_score + w_sph * spherical_score
    
    # Ranked Probability Score (RPS) for ordered 'score' scales
    is_score = (qtype == 1).float()
    if is_score.any():
        k = mask.sum(-1).clamp(min=2).float()
        cdf_p = torch.cumsum(probs, -1)
        cdf_t = torch.cumsum(target, -1)
        rps = (((cdf_p - cdf_t) ** 2) * mask).sum(-1) / (k - 1)
        reward = reward - w_rps * rps * is_score
        
    return -reward.mean()  # Minimize negative proper reward
